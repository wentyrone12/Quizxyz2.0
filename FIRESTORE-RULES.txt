WHITE_QUIZXYZ 2.3 - Firestore Rules

Paste into Firebase Console > Firestore Database > Rules.
The /quizzes collection is public-read for shared decks. The /mail collection is a protected write queue for Firebase Trigger Email (configure the extension separately).

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /quizzes/{quizId} {
      allow read: if true;
      allow create: if request.auth != null
        && request.resource.data.ownerUid == request.auth.uid;
      allow update, delete: if false;
    }

    match /mail/{mailId} {
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.to == request.auth.token.email;
      allow read, update, delete: if false;
    }
  }
}
